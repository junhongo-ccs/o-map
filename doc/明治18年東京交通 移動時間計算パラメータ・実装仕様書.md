### 所要時間計算実装コード例

#### JavaScript 版モジュール

```javascript
/**
 * 明治18年（1885年）東京交通時間計算パラメータ
 */
const MEIJI18_TRANSIT_PARAMS = {
  // 徒歩（成人男性）：1日10里（40km）/ 日中約10.5時間歩行ベース
  walk_male: {
    name: "徒歩（成人男性）",
    speedKmh: 3.8,
    waitTimeMin: 0
  },
  // 徒歩（一般・女性）：1日7〜8里（28〜32km）ベース
  walk_female: {
    name: "徒歩（一般・女性）",
    speedKmh: 2.8,
    waitTimeMin: 0
  },
  // 東京馬車鉄道：新橋〜浅草（万世橋経由42分/浅草橋経由46分）、多客時1分間隔・自由乗降
  horse_tram: {
    name: "東京馬車鉄道",
    speedKmh: 9.0,
    waitTimeMin: 1.5
  },
  // 官設鉄道（新橋–横浜）：29kmを約55分、1日13往復（約75分間隔）
  government_rail: {
    name: "官設鉄道（新橋–横浜）",
    speedKmh: 30.8,
    waitTimeMin: 37.5 // 平均待ち時間（75分 / 2）
  },
  // 日本鉄道 品川線：新宿–赤羽 32分、新橋–赤羽 75分（1日3往復）
  shinagawa_line: {
    name: "日本鉄道 品川線",
    speedKmh: 30.0,
    sectionTimesMin: {
      shinbashi_akabane: 75,
      shinjuku_akabane: 32 // 官報・便覧表の一次資料検証値
    }
  }
};

/**
 * 移動時間の計算（距離 ÷ 速度 ＋ 待ち時間）
 * @param {number} distanceKm - 移動距離 (km)
 * @param {string} modeKey - 交通手段のキー ('walk_male', 'horse_tram', etc.)
 * @returns {object} 計算結果オブジェクト
 */
function calculateTravelTime(distanceKm, modeKey) {
  const mode = MEIJI18_TRANSIT_PARAMS[modeKey];
  if (!mode) throw new Error("指定された交通手段が存在しません");

  const moveTimeMin = (distanceKm / mode.speedKmh) * 60;
  const waitTimeMin = mode.waitTimeMin ?? 0;
  const totalTimeMin = moveTimeMin + waitTimeMin;

  return {
    modeName: mode.name,
    distanceKm: distanceKm,
    moveTimeMin: Math.round(moveTimeMin),
    waitTimeMin: Math.round(waitTimeMin),
    totalTimeMin: Math.round(totalTimeMin)
  };
}

// 使用例
console.log(calculateTravelTime(3.0, 'horse_tram')); 
// Output: { modeName: '東京馬車鉄道', distanceKm: 3, moveTimeMin: 20, waitTimeMin: 2, totalTimeMin: 22 }
```

#### Python 版クラス構成

```python
from dataclasses import dataclass
from typing import Dict, Optional

@dataclass
class TransitMode:
    name: str
    speed_kmh: float
    wait_time_min: float

class Meiji18TravelCalculator:
    MODES: Dict[str, TransitMode] = {
        "walk_male": TransitMode("徒歩（成人男性）", 3.8, 0.0),
        "walk_female": TransitMode("徒歩（一般・女性）", 2.8, 0.0),
        "horse_tram": TransitMode("東京馬車鉄道", 9.0, 1.5),
        "gov_rail": TransitMode("官設鉄道（新橋–横浜）", 30.8, 37.5),
    }

    @classmethod
    def calculate(cls, distance_km: float, mode_key: str) -> dict:
        mode = cls.MODES.get(mode_key)
        if not mode:
            raise ValueError(f"Unknown mode: {mode_key}")
        
        move_time = (distance_km / mode.speed_kmh) * 60
        total_time = move_time + mode.wait_time_min
        
        return {
            "mode": mode.name,
            "distance_km": distance_km,
            "move_time_min": round(move_time),
            "wait_time_min": round(mode.wait_time_min),
            "total_time_min": round(total_time)
        }

# 使用例
# result = Meiji18TravelCalculator.calculate(5.0, "walk_male")
```
